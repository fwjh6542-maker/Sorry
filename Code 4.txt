<script>
// Function to create falling heart emojis
function createFallingHeart() {
const hearts = ['❤️', '💜'];
const element = document.createElement('div');
element.className = 'heart-emoji';
element.innerHTML = hearts[Math.floor(Math.random() * hearts.length)];
element.style.left = Math.random() * 100 + 'vw';
element.style.animationDuration = (Math.random() * 3 + 3) + 's';
document.body.appendChild(element);

element.addEventListener('animationend', () => element.remove());
}

// Create falling hearts periodically
setInterval(() => createFallingHeart(), 200);
